#include <tunables/global>

profile mcp_ha flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities needed by s6 / bashio bootstrap and the privilege drop
  capability chown,
  capability dac_override,
  capability setgid,
  capability setuid,
  capability kill,

  signal (send) set=(kill,term,int,quit,cont),

  # S6-Overlay init tree
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /usr/lib/bashio/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/s6/** rix,
  /run/{,**} rwk,
  /dev/tty rw,
  /tmp/** rwk,

  # Add-on data and entrypoint
  /data/** rw,
  /run.sh r,
  /app/** r,

  # The Node server runs confined in a child profile
  /usr/local/bin/node cx -> node,

  profile node flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>
    #include <abstractions/nameservice>

    # MCP server socket and Home Assistant API access
    network tcp,
    network udp,

    /usr/local/bin/node mr,
    /app/** r,
    /data/** rw,
    /etc/hosts r,
    /etc/resolv.conf r,
    /etc/ssl/** r,
    /proc/@{pid}/** r,
    /proc/sys/** r,
    /sys/fs/cgroup/** r,
    /dev/urandom r,
    /dev/null rw,
    /tmp/** rwk,
  }
}
